\usection{Objetivos}

\usubsection{Objetivo General}
Desarrollar un lenguaje de programación visual y su entorno basado en flujos de datos para fomentar el desarrollo del pensamiento computacional en niños y jóvenes.

\usubsection{Objetivos Específicos}

\newcommand{\oeOne}{Analizar la problemática para caracterizar un lenguaje de programación visual bajo el paradigma de flujo de datos orientado a niños y jóvenes para desarrollar el pensamiento computacional, así como su entorno de desarrollo y ejecución.}
\newcommand{\oeTwo}{Diseñar la sintaxis visual y la semántica formal del lenguaje de programación visual, y el entorno de desarrollo y ejecución, según el análisis realizado.}
\newcommand{\oeThree}{Implementar el entorno de desarrollo visual y el motor de ejecución, en función del diseño realizado.}
\newcommand{\oeFour}{Validar el lenguaje de programación visual desarrollado, con respecto al análisis realizado.}
